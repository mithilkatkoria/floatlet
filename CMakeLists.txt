cmake_minimum_required(VERSION 3.24)
project(Floatlet VERSION 0.6.2 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(model_tests tests/model_tests.cpp)
target_include_directories(model_tests PRIVATE src)
add_test(NAME model COMMAND model_tests)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Use MSVC and Windows SDK 10.0.26100 or newer")
  endif()
  add_executable(DelightIsland WIN32 src/app/resources.rc src/app/main.cpp src/ui/renderer.cpp src/platform/shelf.cpp src/services/media.cpp src/storage/settings.cpp)
  set_target_properties(DelightIsland PROPERTIES OUTPUT_NAME Floatlet)
  target_include_directories(DelightIsland PRIVATE src)
  target_compile_definitions(DelightIsland PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(DelightIsland PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_libraries(DelightIsland PRIVATE windowsapp CoreMessaging d3d11 dxgi d2d1 dwrite ole32 shell32 shlwapi shcore user32 gdi32 advapi32 wtsapi32 comctl32 uuid wbemuuid oleaut32 wlanapi winhttp crypt32)
  target_link_options(DelightIsland PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app/app.manifest)
  set_property(TARGET DelightIsland model_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  install(TARGETS DelightIsland RUNTIME DESTINATION .)
